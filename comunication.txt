local m = {}
_G.console = {}

function async(fn)
    if not fn then io.stderr:write("attempt to def async with nil\n") end
    local fn = string.dump(fn)
    if fn then
        function call(...)
            local vars={...}
            for i,v in ipairs(vars) do
                os.setenv(i,v)
            end
            local fun = fn
            local self = {}
            self.id = tostring(math.random(100000,999999))
            local tmp = os.getenv("HOME").."/"..self.id..".lua"
            local f = io.open(tmp,"wb")
            if f then
                f:write(fun)
                f:close()
                os.execute("lua "..tmp.." &")
            else
                print("error creating file")
            end
        end
        return call
    end
end

function console.readkey()
    os.execute("stty -icanon -echo -isig min 1 time 0")
    local k = io.read(1)
    os.execute("stty sane")
    return k
end

function console.readasync()
    os.execute("stty -icanon -echo -isig min 0 time 0")
    local inp = io.read()
    os.execute("stty sane")
    return inp
end

function console.readkeyasync()
    os.execute("stty -icanon -echo -isig min 0 time 0")
    local k = io.read(1)
    os.execute("stty sane")
    return k
end

function console.getmouse()
    os.execute("stty -icanon -echo -isig min 1 time 0")
    io.write("\27[?1006h")
    local k = io.read(1)
    local g = nil
    local mouse = nil
    if k == "\27" then
        g = k..io.read(10)
        local a,b,c = g:match("(%d*);(%d*);(%d*)")
        a = tonumber(a)
        b = tonumber(b)
        c = tonumber(c)
        mouse = {
            button = a,
            x = b,
            y = c
        }
    end
    os.execute("stty sane")
    io.write("\27[?1006l")
    return mouse
end
